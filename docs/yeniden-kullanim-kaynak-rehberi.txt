1.1 PBKDF2-SHA256 Parola Hashing (OWASP)
--------------------------------------------------------------------------------
Dosya: DbHelper.cs -> HashPassword / VerifyPassword
NuGet: yok (System.Security.Cryptography dahili)

KOD:
  public static string HashPassword(string password)
  {
      byte[] salt = new byte[16];
      using (var rng = new RNGCryptoServiceProvider()) rng.GetBytes(salt);
      using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, 100000))
      {
          byte[] hash = pbkdf2.GetBytes(32);
          return $"PBKDF2:{Convert.ToBase64String(salt)}:{Convert.ToBase64String(hash)}";
      }
  }

  public static bool VerifyPassword(string password, string stored, out bool needsMigration)
  {
      needsMigration = false;
      if (stored.StartsWith("PBKDF2:"))
      {
          var parts = stored.Split(':');
          byte[] salt = Convert.FromBase64String(parts[1]);
          byte[] hash = Convert.FromBase64String(parts[2]);
          using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, 100000))
          {
              byte[] computed = pbkdf2.GetBytes(32);
              return CryptographicEquals(computed, hash); // sabit zamanli karsilastirma
          }
      }
      // eski duz metin / SHA-256 -> needsMigration = true, ilk giriste yukselt
      needsMigration = true;
      return stored == password; // veya eski SHA-256 dogrulama
  }

  private static bool CryptographicEquals(byte[] a, byte[] b)
  {
      if (a.Length != b.Length) return false;
      int result = 0;
      for (int i = 0; i < a.Length; i++) result |= a[i] ^ b[i];
      return result == 0;
  }

YENIDEN KULLANIM:
  - Her projede DbHelper veya SecurityHelper sinifina koyun.
  - Rfc2898DeriveBytes .NET Framework 4.8'de mevcut.
  - ASLA MD5 / SHA1 / tek tur SHA-256 kullanmayin.
  - Geriye donuk uyumluluk: eski hash'li hesaplar ilk basarili giriste otomatik PBKDF2'ye yukseltilir.

3.1 Repository Pattern + Unit of Work
--------------------------------------------------------------------------------
Dosyalar: Data\IUnitOfWork.cs, Data\UnitOfWork.cs, Data\*Repository.cs

  public interface IUnitOfWork : IDisposable
  {
      ProductRepository Products { get; }
      CustomerRepository Customers { get; }
      OrderRepository Orders { get; }
      void Commit();
  }

  public class UnitOfWork : IUnitOfWork
  {
      private readonly SqlConnection _conn;
      private SqlTransaction _trans;
      public ProductRepository Products { get; }
      public CustomerRepository Customers { get; }

      public UnitOfWork(string connStr)
      {
          _conn = new SqlConnection(connStr);
          _conn.Open();
          _trans = _conn.BeginTransaction();
          Products = new ProductRepository(_conn, _trans);
          Customers = new CustomerRepository(_conn, _trans);
      }
      public void Commit() { _trans.Commit(); _trans = _conn.BeginTransaction(); }
      public void Dispose() { _trans?.Rollback(); _conn?.Dispose(); }
  }

Kullanim:
  using (var uow = new UnitOfWork(connStr))
  {
      uow.Products.Update(p);
      uow.Commit();
  }
  // commit cagrilmazsa otomatik rollback

3.2 Async DAL (UI Donmaz)
--------------------------------------------------------------------------------
Dosya: DbHelper.cs -> ExecuteQueryAsync, ExecuteNonQueryAsync, ExecuteScalarAsync

  public static async Task<DataTable> ExecuteQueryAsync(string sql, SqlParameter[] pars, CancellationToken ct = default)
  {
      using (var conn = new SqlConnection(ConnectionString))
      using (var cmd = new SqlCommand(sql, conn))
      {
          if (pars != null) cmd.Parameters.AddRange(pars);
          await conn.OpenAsync(ct).ConfigureAwait(false);
          using (var reader = await cmd.ExecuteReaderAsync(ct).ConfigureAwait(false))
          {
              var dt = new DataTable();
              dt.Load(reader);
              return dt;
          }
      }
  }

Kural:
  - .ConfigureAwait(false) her async cagrida.
  - UI thread'de async void event handler, ic mantik Task / Task<T>.
  - CancellationToken destegi; iptal durumunda kaynaklar temizlenir.

  public static void ExportToCsv(DataTable dt, string filePath)
  {
      using (var sw = new StreamWriter(filePath, false, new UTF8Encoding(true))) // BOM
      {
          sw.WriteLine(string.Join(";", dt.Columns.Cast<DataColumn>().Select(c => $"\"{c.ColumnName}\"")));
          foreach (DataRow r in dt.Rows)
              sw.WriteLine(string.Join(";", r.ItemArray.Select(v => $"\"{v}\"")));
      }
  }

6.1 WAC (Agirlikli Ortalama Maliyet)
--------------------------------------------------------------------------------
  public static decimal CalculateWAC(decimal currentQty, decimal currentCost,
                                    decimal incomingQty, decimal incomingCost)
  {
      if (currentQty + incomingQty == 0) return incomingCost;
      if (currentQty <= 0) return incomingCost; // negatif stokta yeni fiyata gec
      return ((currentQty * currentCost) + (incomingQty * incomingCost)) / (currentQty + incomingQty);
  }

10. DIKKAT EDILMESI GEREKEN TUZAKLAR
================================================================================

1. Norton/AV false-positive:
   - Test betikleri %TEMP%\antigravity_ims\ altinda calissin.
   - Desktop'a .ps1 yazilmamali.
   - Klasor istisnasi (Settings -> Virus -> Scratch klasoru) + Fidye Yazilimi Koruma izinli uygulama listesine powershell.exe.

2. Encoding:
   - Turkce karakterler icin dosyalar UTF-8 BOM ile kaydedilmeli.
   - PowerShell 5.1 ANSI okur; BOM olmadan Türkçe stringler bozulur.
   - Write tool sonrasi BOM ekle: [System.IO.File]::WriteAllText(path, text, (New-Object System.Text.UTF8Encoding($true))).

3. MSBuild secimi:
   - vswhere SSMS'in MSBuild'ini secebilir.
   - Filtrele: Where-Object { $_ -notmatch "SQL Server Management Studio" }
   - Alternatif: Get-ChildItem "C:\Program Files\Microsoft Visual Studio\2022\*\MSBuild\Current\Bin\MSBuild.exe"

4. Start-Process -ArgumentList:
   - Bosluklu yollar bolunur ("C:\Users\FURKAN" -> hata).
   - Cozum: & powershell.exe -ExecutionPolicy Bypass -File "path" (tek string).

5. vstest.console /nologo gecersiz:
   - Bu parametreyi kullanmayin; sadece /logger:trx;LogFileName=... kullanin.

6. DbHelper statik constructor:
   - DataDirectory ayari uygulama acilisinda yapilmali (LocalDB attach path).
   - Settings.Default.Reload() ile kullanici ayarlari yuklenmeli.

7. Transaction rollback:
   - UnitOfWork.Dispose() commit cagrilmamissa otomatik rollback yapmali.
   - try/catch icinde commit sonrasi Rollback cagrisi unutulmamali.

8. Concurrency:
   - Stok dusumunde AND ProdQty >= @qty kosulu + etkilenen satir kontrolu sart.
   - ROWLOCK, UPDLOCK hint'leri kullanilmali.

9. Norton istisnasi yapilmadan test calistirma:
   - Antivirus kapatilmamali; klasor istisnasi + izinli uygulama listesi yeterli.

10. Git binary kalintilari:
    - .pfx ve buyuk jpg dosyalari repodan silinmeli.
    - Silme sirasi: Resources.resx data entry -> Resources.Designer.cs property -> .csproj satiri -> sonra dosya silme.
    - Aradaki derleme dogrulamasi zorunlu (yoksa build kirilir).

11. Prompt siralamasi:
    - Antigravity'ye her seferinde YALNIZCA bir prompt ver.
    - Her prompt sonrasi derle (Ctrl+Shift+B), manuel smoke test yap, git commit et.
    - P0 -> P1 -> P2 -> P3 sirasina uy.
